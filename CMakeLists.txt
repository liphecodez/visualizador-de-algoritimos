cmake_minimum_required(VERSION 3.20)
project(algorithm_visualizer C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
include(CTest)
function(enable_project_warnings target)
  # Keep project code in strict C99; Raylib's web backend needs GNU extensions.
  set_target_properties(${target} PROPERTIES C_EXTENSIONS OFF)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  elseif(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  endif()
endfunction()
set(ALGORITHMS algoritimos/bubblesort.c algoritimos/selectionsort.c algoritimos/insertionsort.c algoritimos/mergesort.c algoritimos/quicksort.c)
add_library(sorting STATIC ${ALGORITHMS})
target_include_directories(sorting PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
enable_project_warnings(sorting)
option(BUILD_VISUALIZER "Build Raylib UI" ON)
if(BUILD_TESTING)
  add_executable(test_sort tests/test_sort.c)
  target_link_libraries(test_sort PRIVATE sorting)
  enable_project_warnings(test_sort)
  if(EMSCRIPTEN)
    target_link_options(test_sort PRIVATE "-sENVIRONMENT=node" "-sASSERTIONS=2" "-sSAFE_HEAP=1")
  endif()
  add_test(NAME sorting COMMAND test_sort)
endif()
if(BUILD_VISUALIZER)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  # Raylib 5.5: immutable source revision and verified archive checksum.
  FetchContent_Declare(raylib
    URL https://codeload.github.com/raysan5/raylib/tar.gz/c1ab645ca298a2801097931d1079b10ff7eb9df8
    URL_HASH SHA256=be3443c3fb97f7cbd18ae3f0ea2a73e6c1486128df33b639dce7e44070fc285d)
  FetchContent_MakeAvailable(raylib)
  add_executable(visualizador main.c gerar-dados.c)
  target_link_libraries(visualizador PRIVATE sorting raylib)
  enable_project_warnings(visualizador)
  if(EMSCRIPTEN)
    target_compile_definitions(visualizador PRIVATE PLATFORM_WEB)
    set_target_properties(visualizador PROPERTIES OUTPUT_NAME index SUFFIX ".html" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist/web")
    target_link_options(visualizador PRIVATE "--shell-file=${CMAKE_SOURCE_DIR}/web/shell.html" "-sUSE_GLFW=3" "-sALLOW_MEMORY_GROWTH=1" "-sENVIRONMENT=web" "-sASSERTIONS=1")
    file(MAKE_DIRECTORY "${CMAKE_SOURCE_DIR}/dist/web")
    configure_file("${raylib_SOURCE_DIR}/LICENSE" "${CMAKE_SOURCE_DIR}/dist/web/LICENSE-raylib.txt" COPYONLY)
    configure_file("${CMAKE_SOURCE_DIR}/web/.nojekyll" "${CMAKE_SOURCE_DIR}/dist/web/.nojekyll" COPYONLY)
  endif()
endif()
